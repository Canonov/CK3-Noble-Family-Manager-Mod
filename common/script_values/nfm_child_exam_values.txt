# Approximation of send_child_to_examination_interaction's duel weights.

nfm_value_child_exam_success_weight = {
    value = 40

    add = {
        value = learning
        subtract = medium_skill_rating
        multiply = 3.5
        min = -49
    }

    if = {
        limit = {
            OR = {
                scope:nfm_child_exam_actor ?= {
                    employs_court_position = court_tutor_court_position
                }
                liege ?= {
                    employs_court_position = court_tutor_court_position
                }
            }
        }
        add = 20
    }

    if = {
        limit = {
            house ?= {
                has_house_aspiration_parameter = examination_success_bonus
            }
        }
        add = 10
    }

    if = {
        limit = { learning >= 10 }
        add = learning
    }

    min = 5
}

nfm_value_child_exam_failure_weight = {
    value = 60

    add = {
        value = learning
        subtract = medium_skill_rating
        multiply = -0.5
    }
}

nfm_value_child_exam_pass_chance = {
    value = nfm_value_child_exam_success_weight
    divide = {
        value = nfm_value_child_exam_success_weight
        add = nfm_value_child_exam_failure_weight
        min = 1
    }
    multiply = 100
}
